```python
from dataclasses import dataclass, field
from abc import ABC, abstractmethod
from typing import Optional

@dataclass
class OpportunityRaw:
    title: str
    company: str
    location: str
    source_url: str
    source: str
    source_type: str = "ats"
    salary_usd_low: Optional[float] = None
    salary_usd_high: Optional[float] = None
    raw_payload: dict = None

@dataclass
class SourceResult:
    source_name: str
    opportunities: list[OpportunityRaw]
    errors: list[str] = field(default_factory=list)
    rate_limited: bool = False
    fetch_duration_sec: float = 0.0

class BaseSource(ABC):
    name: str = "base"
    @abstractmethod
    def fetch(self, constraints: dict) -> SourceResult: ...
    def _is_enabled(self, config: dict) -> bool: ...
    def _rate_limited_sleep(self) -> None: ...
    def _exponential_backoff(self, attempt: int, base_sec: float = 2.0) -> None: ...
```

    AshbySource(BaseSource):
    - name = "ashby"
    - __init__(self, config: dict):
        board_name = config.get("board_name", "")
        company_name = config.get("company_name", "")
        include_compensation = config.get("include_compensation", True)
        enabled = config.get("enabled", False)
        url = f"https://api.ashbyhq.com/posting-api/job-board/{board_name}"
        if include_compensation: url += "?includeCompensation=true"

    - fetch(constraints) -> SourceResult:
        Return empty + error if board_name blank.
        GET self.url, timeout=30.
        429 → SourceResult(rate_limited=True, errors=["Rate limited (429)..."]).
        raise_for_status() for other 4xx/5xx → catch HTTPError → return error.
        Parse data["jobPostings"] (list).
        For each posting:
          salary_usd_low = posting.get("compensation",{}).get("salary",{}).get("min")
          salary_usd_high = posting.get("compensation",{}).get("salary",{}).get("max")
          currency = posting.get("compensation",{}).get("salary",{}).get("currency","USD")
          Only use salary if currency == "USD" (else leave None).
          location = posting.get("location","")
          company = self.company_name (Ashby doesn't return company in job; use config)
          source_url = posting.get("url","")
          raw_payload = posting (full dict for audit)
        On Timeout → return error "Request timeout: ...".
        On any other Exception → return error "Unexpected error: ...".
        NEVER raise from fetch().

    Salary confidence tagging is NOT done in the connector (handled by orchestrator in 02-04).
    The fact that ashby returns salary_usd_low/high is sufficient for the orchestrator
    to tag confidence=HIGH.
  </behavior>
