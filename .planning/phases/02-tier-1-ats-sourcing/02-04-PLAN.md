    stages/fetch.py must implement:

    infer_remote_status(location: str, remote_policy: str = None) -> str:
      If remote_policy in ["fully_remote","hybrid_remote_preferred","hybrid_onsite_required","onsite_only"]
        → return remote_policy directly.
      Else check location_lower:
        "remote" in location_lower → "fully_remote"
        "hybrid" in location_lower → "hybrid_remote_preferred"
        else → "onsite_only"

    _infer_salary_confidence(source: str, has_salary: bool) -> str:
      source in ["greenhouse","ashby"] and has_salary → "HIGH"
      else → "LOW"

    _infer_salary_evidence_type(source: str, has_salary: bool) -> str:
      has_salary and source in ["greenhouse","ashby"] → "employer_posted"
      not has_salary → "not_disclosed"
      else → "not_disclosed"

    normalize_opportunity(raw: OpportunityRaw, run_id: str) -> dict:
      Returns a dict with ALL DATA-01 schema fields.
      access_date = datetime.utcnow().isoformat() + "Z"
      observed_at = access_date (same timestamp for this run)
      opportunity_id = str(uuid.uuid4())
      title = normalize_title(raw.title)
      company = normalize_company(raw.company)
      location = normalize_location(raw.location)
      remote_status = infer_remote_status(raw.location, raw.raw_payload.get("remotePolicy") if raw.raw_payload else None)
      salary_confidence = _infer_salary_confidence(raw.source, bool(raw.salary_usd_low))
      salary_evidence_type = _infer_salary_evidence_type(raw.source, bool(raw.salary_usd_low))
      fit_score = 0, growth_score = 0, confidence = "LOW", tags = []
      lane = "Remote USD"
      data_quality = "confirmed"
      run_id = run_id

    _load_ats_config() -> dict:
      Load config_sources.yaml from MODULE_ROOT / "radar" / "config_sources.yaml".
      Use import yaml (yaml is stdlib via PyYAML which is already in requirements.txt).
      If file not found, return minimal disabled config (all four platforms, enabled=False, boards=[]).
      MODULE_ROOT = Path(__file__).resolve().parent.parent.parent  (stages/ -> radar/ -> TARIQ__career_radar/)

    run_fetch(constraints: dict, run_id: str) -> dict:
      Load config via _load_ats_config().
      Build source instances for each ENABLED board in the config:
        For greenhouse: one GreenhouseSource per board in config["tier_1_ats"]["greenhouse"]["boards"]
        For lever: one LeverSource per board
        For ashby: one AshbySource per board (pass include_compensation from board config)
        For workable: one WorkableSource per board
        Skip entire platform if platform-level enabled=False.
      
      Sequential fetch loop (NOT parallel — stagger rate limiting):
        For each source instance:
          try:
            result = source.fetch(constraints)  # already safe; never raises
            # Normalize each raw opportunity
            for raw_opp in result.opportunities:
              normalized = normalize_opportunity(raw_opp, run_id)
              all_opportunities.append(normalized)
            # Log errors (if any) without aborting
            if result.errors:
              blocked_sources.append({
                "source": source.name,
                "board": getattr(source, "board_token", None) or
                         getattr(source, "site", None) or
                         getattr(source, "board_name", None) or
                         getattr(source, "account_subdomain", ""),
                "errors": result.errors,
                "rate_limited": result.rate_limited,
              })
              logger.warning(f"{source.name}: errors={result.errors}")
            else:
              logger.info(f"{source.name}: fetched {len(result.opportunities)} opportunities")
          except Exception as e:
            # Defensive: connectors should never raise, but catch anyway
            blocked_sources.append({
              "source": source.name,
              "board": "",
              "errors": [f"Unhandled exception: {type(e).__name__}: {e}"],
              "rate_limited": False,
            })
            logger.exception(f"{source.name} crashed unexpectedly")
      
      fetch_summary = {
        "total_fetched": len(all_opportunities),
        "total_blocked_sources": len(blocked_sources),
        "run_result": (
          "success" if not blocked_sources
          else "partial_success" if all_opportunities
          else "failure"
        ),
      }
      
      return {
        "opportunities": all_opportunities,
        "blocked_sources": blocked_sources,
        "fetch_summary": fetch_summary,
      }
