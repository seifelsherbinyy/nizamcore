    ```python
    @dataclass
    class ScoreBreakdown:
        fit: int = 0
        salary_upside: int = 0
        growth: int = 0
        visa_feasibility: int = 0
        company_strength: int = 0
        referral_leverage: int = 0
        freshness: int = 0

        def total_penalty(self) -> int:
    ```

    compute_fit_score(opportunity: dict, profile: dict) -> int:
    - Extract title + description, join to role_text (lowercase)
    - Flatten profile["role_keywords"] (handles both list and dict[str, list] format)
    - Count keyword substring matches in role_text
    - score = int((matches / total_keywords) * 100) if total_keywords > 0 else 0
    - Return min(100, max(0, score))
    - Guard: if role_text empty or no keywords → return 0

    compute_salary_upside_score(opportunity: dict) -> int:
    - salary = opportunity.get("salary_usd_high")
    - if salary is None: return 0
    - min_ref = SALARY_THRESHOLDS["min_ref"]  # 60000
    - high_threshold = SALARY_THRESHOLDS["high_threshold"]  # 150000
    - raw = max(0, min(100, int((salary - min_ref) * 100 // (high_threshold - min_ref))))
    - mult = SALARY_CONFIDENCE_MULTIPLIER.get(opportunity.get("salary_confidence", "LOW"), 50)
    - return int(raw * mult // 100)

    compute_growth_score(opportunity: dict) -> int:
    - category = opportunity.get("role_category", "").upper()
    - company = opportunity.get("company", "").lower()
    - base = GROWTH_BASE_HIGH if category in GROWTH_CATEGORIES else GROWTH_BASE_LOW
    - if any(t in company for t in TIER1_COMPANIES): base = min(100, base + TIER1_BOOST)
    - return base

    compute_visa_feasibility_score(opportunity: dict) -> int:
    - visa = opportunity.get("visa_feasibility", "visa_sponsored_unclear")
    - return VISA_SCORE_MAP.get(visa, 50)

    compute_company_strength_score(opportunity: dict) -> int:
    - company = opportunity.get("company", "").lower()
    - description = opportunity.get("description", "").lower()
    - text = f"{company} {description}"
    - if any(t in company for t in TIER1_COMPANIES): return 90
    - if "series b" in text or "series c" in text or "series d" in text: return 70
    - if "profitable" in text or ("growing" in text and "startup" not in text): return 60
    - if "startup" in text and "series" not in text: return 40
    - return 50

    compute_referral_leverage_score(opportunity: dict) -> int:
    - source_type = opportunity.get("source_type", "").lower()
    - source = opportunity.get("source", "").lower()
    - if source_type == "ats" or any(ats in source for ats in ATS_PLATFORMS): return 70
    - if any(sp in source for sp in SIDE_INCOME_PLATFORMS): return 80
    - if source_type in {"rss_feed", "manual"}: return 50
    - return 50

    compute_freshness_score(opportunity: dict, now: datetime.datetime | None = None) -> int:
    - if now is None: now = datetime.datetime.now(datetime.timezone.utc)
    - access_str = opportunity.get("access_date")
    - if not access_str: return FRESHNESS_SCORES["unknown"]
    - Parse: use datetime.datetime.fromisoformat(access_str.replace("Z", "+00:00"))
    - Ensure now is timezone-aware: if now.tzinfo is None, assume UTC
    - age_days = (now - access_date).days
    - if age_days <= FRESHNESS_FRESH_DAYS: return FRESHNESS_SCORES["fresh"]
    - elif age_days <= FRESHNESS_AGING_DAYS: return FRESHNESS_SCORES["aging"]
    - else: return FRESHNESS_SCORES["stale"]

    compute_side_income_score(opportunity: dict) -> int:
    - source = opportunity.get("source", "").lower()
    - company = opportunity.get("company", "").lower()
    - is_side = any(sp in source or sp in company for sp in SIDE_INCOME_PLATFORMS)
    - return SIDE_INCOME_SCORE if is_side else 0

    ScoringEngine class:
    - Class attribute: WEIGHTS = WEIGHTS  (from scoring_config — this is what tests assert)
    - __init__(self, profile: dict | None = None): self.profile = profile or {}
    - score(self, opportunity: dict, now: datetime.datetime | None = None) -> tuple[int, ScoreBreakdown]:
        1. Build ScoreBreakdown with all 8 dimensions (call compute_* functions)
        2. Compute base_score using integer arithmetic with WEIGHTS_INT
        3. Call _compute_penalties() → penalties dict
        4. breakdown.penalties = penalties
        5. final_score = max(0, min(100, base_score - sum(penalties.values())))
        6. return final_score, breakdown
    
    - _compute_penalties(self, opportunity: dict, breakdown: ScoreBreakdown) -> dict[str, int]:
        ALL 5 checks are INDEPENDENT (no early returns, no cascade).
        Compute each check, collect into penalties dict, return dict.
        
        Check 1 (scam_risk):
        text = (title + company + description).lower()
        if any(kw in text for kw in SCAM_KEYWORDS): penalties["scam_risk"] = PENALTY_VALUES["scam_risk"]

        conf = opportunity.get("salary_confidence", "LOW")
        desc = opportunity.get("description", "").lower()

            penalties["unclear_pay"] = PENALTY_VALUES["unclear_pay"]
        
        Check 3 (severe_skill_mismatch):
        avoid = self.profile.get("avoid_flags", [])
        role_cat = opportunity.get("role_category", "")
        if breakdown.fit < 30 and role_cat in avoid:
            penalties["severe_skill_mismatch"] = PENALTY_VALUES["severe_skill_mismatch"]
        
        Check 4 (exploitative_unpaid):
        salary_high = opportunity.get("salary_usd_high")
        title_desc = (opportunity.get("title","") + " " + opportunity.get("description","")).lower()
        if (salary_high is not None and salary_high == 0) or any(kw in title_desc for kw in UNPAID_KEYWORDS):
            penalties["exploitative_unpaid"] = PENALTY_VALUES["exploitative_unpaid"]
        
        return penalties
