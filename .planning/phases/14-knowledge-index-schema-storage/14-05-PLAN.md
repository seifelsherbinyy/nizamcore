    import pytest
    from HIKMAH__knowledge_index.index.schema import (
        validate_index_schema, VALID_PERSONAS, CONTEXT_TAGS_WHITELIST
    )
    
    class TestSchemaValidation:
        """Test index schema validation logic."""
        
        def test_valid_index_passes(self, sample_index_dict):
            """Complete valid index should pass validation."""
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is True
            assert error is None
        
        def test_missing_version(self, sample_index_dict):
            """Index missing 'version' field should fail."""
            del sample_index_dict["version"]
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is False
            assert "version" in error.lower()
        
        def test_missing_persona(self, sample_index_dict):
            """Index missing 'persona' field should fail."""
            del sample_index_dict["persona"]
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is False
            assert "persona" in error.lower()
        
        def test_invalid_version_format(self, sample_index_dict):
            """Version must be 1.x format; 2.0 should fail in v1.1."""
            sample_index_dict["version"] = "2.0"
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is False
            assert "version" in error.lower()
        
        def test_unknown_persona(self, sample_index_dict):
            """Unknown persona should fail."""
            sample_index_dict["persona"] = "UNKNOWN_PERSONA"
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is False
            assert "unknown" in error.lower() or "persona" in error.lower()
        
        def test_all_valid_personas(self, valid_personas):
            """All 11 personas should be valid."""
            for persona in valid_personas:
                index = {
                    "version": "1.0",
                    "persona": persona,
                    "initialized_at": "2026-06-20T14:30:00Z",
                    "last_updated": "2026-06-20T14:30:00Z",
                    "topics": [],
                    "completions": [],
                    "activity_history": [],
                    "stalled_work": [],
                    "context_snapshots": [],
                    "metadata": {"source": "test", "locale": "Egypt/Cairo", "language": "en"}
                }
                valid, error = validate_index_schema(index)
                assert valid is True, f"Persona {persona} should be valid"
        
        def test_context_tag_whitelist_enforcement(self, sample_index_dict):
            """Only whitelisted context_tags allowed."""
            sample_index_dict["topics"].append({
                "id": "test-id",
                "name": "Test topic",
                "status": "active",
                "created_at": "2026-06-20T14:30:00Z",
                "last_activity": "2026-06-20T14:30:00Z",
                "context_tags": ["invalid_tag"],  # Not in whitelist
                "confidence": 0.9,
                "key_accomplishments": [],
                "blockers": [],
                "notes": ""
            })
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is False
            assert "context_tag" in error.lower() or "invalid" in error.lower()
        
        def test_no_raw_pii_tags(self, sample_index_dict):

            pii_tags = ["seif", "my_health", "family_name", "financial_amount"]
            for bad_tag in pii_tags:

                    "status": "active",

                    "key_accomplishments": [],
                    "blockers": [],
                    "notes": ""
                })
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is True

            assert sample_index_dict["topics"] == []
            assert sample_index_dict["completions"] == []
            valid, error = validate_index_schema(sample_index_dict)
            assert valid is True
